function(JE_write_gem directory name dependencies nested)
    file(MAKE_DIRECTORY "${JE_FIXTURE_ROOT}/${directory}")
    file(WRITE "${JE_FIXTURE_ROOT}/${directory}/gem.json"
        "{\"gem_name\":\"${name}\",\"dependencies\":[${dependencies}],\"external_subdirectories\":[${nested}]}\n")
endfunction()
JE_write_gem(EngineSeed EngineSeed "" "")
JE_write_gem(Selected Selected "\"Transitive\"" "")
JE_write_gem(Transitive Transitive "" "\"Nested\",\"NativeTools\"")
JE_write_gem(Transitive/Nested Nested "" "")
JE_write_gem(Family Family "\"Middle\"" "\"Middle\"")
JE_write_gem(Family/Middle Middle "\"Leaf\"" "\"Leaf\"")
JE_write_gem(Family/Middle/Leaf Leaf "" "")
JE_write_gem(Unused Unused "" "\"UnusedTools\"")
JE_write_gem(NvCloth NvCloth "" "")
JE_write_gem(ProjectExplicit ProjectExplicit "" "")

    "${JE_FIXTURE_ROOT}/Transitive/NativeTools" "${JE_FIXTURE_ROOT}/Unused/UnusedTools"
    "${JE_FIXTURE_ROOT}/Project/ProjectTools")
if(JE_CASE STREQUAL "missing")
    JE_write_gem(Selected Selected "\"MissingRequired\"" "")

    JE_write_gem(Transitive Transitive "\"Selected\"" "\"Nested\",\"NativeTools\"")

    JE_write_gem(Family Family "" "\"Middle\"")
    JE_write_gem(Family/Middle Middle "" "\"Leaf\"")
    JE_write_gem(Family/Middle/Leaf Leaf "\"Middle\"" "")
endif()

function(JE_require_path list_name relative present)
    set(JE_PATH "${JE_FIXTURE_ROOT}/${relative}")
    file(REAL_PATH "${JE_PATH}" JE_PATH)
    cmake_path(NORMAL_PATH JE_PATH)
    if(present AND NOT JE_PATH IN_LIST ${list_name})
        message(FATAL_ERROR "JE required ${list_name} directory missing: ${relative}")
    elseif(NOT present AND JE_PATH IN_LIST ${list_name})
        message(FATAL_ERROR "JE unused registration configured: ${relative}")
    endif()
endfunction()
JE_require_path(JE_REGISTERED NvCloth TRUE)
JE_require_path(JE_REGISTERED Unused TRUE)
JE_require_path(JE_ACTIVE Unused FALSE)
JE_require_path(JE_ACTIVE Unused/UnusedTools FALSE)
foreach(JE_PATH EngineSeed NativeEngineTools UserTools Project Project/ProjectTools ProjectExplicit
    Family Family/Middle Family/Middle/Leaf)
    JE_require_path(JE_ACTIVE "${JE_PATH}" TRUE)
endforeach()

    JE_require_path(JE_ACTIVE NvCloth TRUE)
else()
    JE_require_path(JE_ACTIVE NvCloth FALSE)
endif()
if(NOT JE_CASE STREQUAL "child_only" AND NOT JE_CASE STREQUAL "child_depends_owner")
    foreach(JE_PATH Selected Transitive Transitive/Nested Transitive/NativeTools)
        JE_require_path(JE_ACTIVE "${JE_PATH}" TRUE)
    endforeach()
    list(FIND JE_ACTIVE "${JE_FIXTURE_ROOT}/Transitive" JE_TRANSITIVE_INDEX)
    list(FIND JE_ACTIVE "${JE_FIXTURE_ROOT}/Selected" JE_SELECTED_INDEX)
    if(NOT JE_TRANSITIVE_INDEX LESS JE_SELECTED_INDEX)
        message(FATAL_ERROR "JE dependent directory must precede its consumer")
    endif()
endif()
if(JE_CASE STREQUAL "child_depends_owner")
    list(FIND JE_ACTIVE "${JE_FIXTURE_ROOT}/Family/Middle" JE_OWNER_INDEX)
    list(FIND JE_ACTIVE "${JE_FIXTURE_ROOT}/Family/Middle/Leaf" JE_CHILD_INDEX)
    if(NOT JE_OWNER_INDEX LESS JE_CHILD_INDEX)
        message(FATAL_ERROR "JE declared owner dependency must precede its child consumer")
    endif()
endif()

    # Recompute the production composition with its existing Dependent:* cache intact.
    set_property(GLOBAL PROPERTY O3DE_ALL_EXTERNAL_SUBDIRECTORIES "")
    get_external_subdirectories_in_use(JE_CACHED)
    if(NOT JE_CACHED STREQUAL JE_ACTIVE)
        message(FATAL_ERROR "JE cached dependency ordering changed the active composition")
    endif()
endif()
