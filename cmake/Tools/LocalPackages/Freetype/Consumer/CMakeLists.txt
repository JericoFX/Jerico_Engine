cmake_minimum_required(VERSION 3.24)

set(JE_FREETYPE_PACKAGE_ROOT "" CACHE PATH "Extracted local Freetype package root")
set(JE_ZLIB_PACKAGE_ROOT "" CACHE PATH "Extracted local zlib package root")
set(JE_FONT_FILE "" CACHE FILEPATH "Existing local font fixture; never packaged or copied")
set(JE_FONT_SHA256 "" CACHE STRING "Audited SHA256 of the local font fixture")
foreach(je_path IN ITEMS JE_FREETYPE_PACKAGE_ROOT JE_ZLIB_PACKAGE_ROOT CMAKE_BINARY_DIR)
    if("${${je_path}}" MATCHES "://")
        message(FATAL_ERROR "${je_path} must not be a URI")
    endif()
    file(TO_CMAKE_PATH "${${je_path}}" je_local_path)
    if(NOT IS_ABSOLUTE "${je_local_path}" OR je_local_path MATCHES "^//")
        message(FATAL_ERROR "${je_path} must be an explicit absolute local path")
    endif()
    cmake_path(NORMAL_PATH je_local_path)
    set(${je_path} "${je_local_path}")
endforeach()
get_filename_component(je_engine_root "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(je_build_root "${je_engine_root}/build")
cmake_path(IS_PREFIX je_build_root "${CMAKE_BINARY_DIR}" NORMALIZE je_inside_build)
if(NOT je_inside_build OR CMAKE_BINARY_DIR STREQUAL je_build_root)
    message(FATAL_ERROR "Consumer binary outputs must be beneath this checkout/build")
endif()
foreach(je_path IN ITEMS JE_FREETYPE_PACKAGE_ROOT JE_ZLIB_PACKAGE_ROOT CMAKE_BINARY_DIR)
    set(je_ancestor "${${je_path}}")
    while(TRUE)
        if(IS_SYMLINK "${je_ancestor}")
            message(FATAL_ERROR "${je_path} must not use symlinks or linked ancestors")
        endif()
        get_filename_component(je_parent "${je_ancestor}" DIRECTORY)
        if(je_parent STREQUAL je_ancestor)
            break()
        endif()
        set(je_ancestor "${je_parent}")
    endwhile()
endforeach()
include("${CMAKE_CURRENT_LIST_DIR}/VerifyFont.cmake")
foreach(je_root IN ITEMS JE_FREETYPE_PACKAGE_ROOT JE_ZLIB_PACKAGE_ROOT)
    if(NOT IS_DIRECTORY "${${je_root}}")
        message(FATAL_ERROR "${je_root} must name an existing local package directory")
    endif()
endforeach()
if(NOT EXISTS "${JE_FREETYPE_PACKAGE_ROOT}/FindFreetype.cmake" OR
   NOT EXISTS "${JE_ZLIB_PACKAGE_ROOT}/FindZLIB.cmake")
    message(FATAL_ERROR "Both explicit package roots must contain their local Find modules")
endif()
project(JE_FreetypeConsumer LANGUAGES CXX)
list(PREPEND CMAKE_MODULE_PATH "${JE_FREETYPE_PACKAGE_ROOT}" "${JE_ZLIB_PACKAGE_ROOT}")
find_package(ZLIB 1.2.11 EXACT REQUIRED MODULE)
find_package(Freetype 2.11.1 EXACT REQUIRED MODULE)
if(NOT TARGET Freetype::Freetype OR NOT TARGET 3rdParty::Freetype OR NOT TARGET ZLIB::ZLIB)
    message(FATAL_ERROR "The packages must preserve their real standard and engine target aliases")
endif()
get_target_property(je_freetype_links Freetype::Freetype INTERFACE_LINK_LIBRARIES)
if(NOT "ZLIB::ZLIB" IN_LIST je_freetype_links)
    message(FATAL_ERROR "The Freetype package must propagate ZLIB::ZLIB to its consumers")
endif()

add_executable(JE_FreetypeGlyphTest FreetypeGlyphTest.cpp)
target_compile_features(JE_FreetypeGlyphTest PRIVATE cxx_std_17)
target_link_libraries(JE_FreetypeGlyphTest PRIVATE 3rdParty::Freetype)
if(MSVC)
    target_compile_options(JE_FreetypeGlyphTest PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME JE.Freetype.LocalGlyph
    COMMAND "${CMAKE_COMMAND}"
        "-DJE_FONT_FILE=${JE_FONT_FILE}"
        "-DJE_FONT_SHA256=${JE_FONT_SHA256}"
        "-DJE_GLYPH_CONSUMER=$<TARGET_FILE:JE_FreetypeGlyphTest>"
        -P "${CMAKE_CURRENT_LIST_DIR}/RunGlyphTest.cmake")
