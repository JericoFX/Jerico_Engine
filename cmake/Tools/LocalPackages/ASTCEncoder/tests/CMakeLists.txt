cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
foreach(JE_PATH IN ITEMS JE_ASTC_PACKAGE_ROOT CMAKE_BINARY_DIR)
    JE_python_require_local_path(${JE_PATH})
endforeach()
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_OUTPUT_IN_BUILD)
if(NOT JE_OUTPUT_IN_BUILD OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE ASTC consumer output must be beneath this checkout/build")
endif()
project(JELocalASTCEncoderConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE ASTC consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_ASTC_PACKAGE_ROOT}")
find_package(astc-encoder 3.2.0 EXACT REQUIRED MODULE)
add_executable(JEASTCEncoderConsumer ASTCEncoderConsumer.cpp)
target_compile_features(JEASTCEncoderConsumer PRIVATE cxx_std_17)
target_compile_options(JEASTCEncoderConsumer PRIVATE /W4)
target_link_libraries(JEASTCEncoderConsumer PRIVATE 3rdParty::astc-encoder)
set_property(TARGET JEASTCEncoderConsumer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
enable_testing()
foreach(JE_CASE IN ITEMS ldr8 ldr16 ldr32 srgb hdr-rgb hdr normal invalid)
    add_test(NAME JE.LocalPackages.ASTC.${JE_CASE} COMMAND JEASTCEncoderConsumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.ASTC.${JE_CASE} PROPERTIES TIMEOUT 120)
endforeach()
add_test(NAME JE.LocalPackages.ASTC.cli COMMAND "${JE_ASTC_PACKAGE_ROOT}/astc-encoder/bin/$<IF:$<CONFIG:Debug>,Debug,Release>/astcenc-sse4.1.exe" -version)
set_tests_properties(JE.LocalPackages.ASTC.cli PROPERTIES TIMEOUT 15 PASS_REGULAR_EXPRESSION "3[.]2[.]0")
