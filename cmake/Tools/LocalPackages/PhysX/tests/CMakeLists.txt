cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_PHYSX_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_IN_BUILD)
if(NOT JE_IN_BUILD OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE PhysX consumer build must be beneath this checkout/build")
endif()
set(CMAKE_CONFIGURATION_TYPES "Debug;Profile;Release" CACHE STRING "Original engine configurations")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
project(JELocalPhysXConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE PhysX consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_PHYSX_PACKAGE_ROOT}")
find_package(PhysX5 5.1.1 EXACT REQUIRED MODULE)
# Check actual imported mapping; the same Profile executable can consume Checked.
get_target_property(JE_PROFILE_LIBRARY PhysX5PhysX_static_64::imported IMPORTED_LOCATION_PROFILE)
if(LY_PHYSX_PROFILE_USE_CHECKED_LIBS)
    set(JE_PROFILE_EXPECTED checked)
else()
    set(JE_PROFILE_EXPECTED profile)
endif()
if(NOT JE_PROFILE_LIBRARY STREQUAL "${JE_PHYSX_PACKAGE_ROOT}/PhysX/physx/bin/static/${JE_PROFILE_EXPECTED}/PhysX_static_64.lib")
    message(FATAL_ERROR "JE PhysX consumer resolved an unexpected Profile archive")
endif()
add_executable(JEPhysXConsumer PhysXConsumer.cpp)
target_compile_features(JEPhysXConsumer PRIVATE cxx_std_17)
target_compile_definitions(JEPhysXConsumer PRIVATE $<$<NOT:$<CONFIG:Debug>>:NDEBUG>)
# Vehicle1 is deprecated in the exact SDK but remains an actual engine consumer.
target_compile_options(JEPhysXConsumer PRIVATE /W4 /WX /wd4996)
target_link_libraries(JEPhysXConsumer PRIVATE 3rdParty::PhysX5)
enable_testing()
foreach(JE_CASE IN ITEMS lifecycle cooking scene character vehicle invalid)
    add_test(NAME JE.LocalPackages.PhysX.${JE_CASE} COMMAND JEPhysXConsumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.PhysX.${JE_CASE} PROPERTIES TIMEOUT 30)
endforeach()
