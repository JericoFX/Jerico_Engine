cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_LZ4_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_BUILD_IS_LOCAL)
if(NOT JE_BUILD_IS_LOCAL OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE lz4 consumer build must be beneath this checkout/build")
endif()
if(NOT EXISTS "${JE_LZ4_PACKAGE_ROOT}/Findlz4.cmake")
    message(FATAL_ERROR "JE_LZ4_PACKAGE_ROOT must contain the extracted JE local Findlz4.cmake")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Test Release archive in both consumer configurations" FORCE)
project(JELocalLz4Consumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE lz4 consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_LZ4_PACKAGE_ROOT}")
find_package(lz4 1.9.4 EXACT REQUIRED MODULE)
get_target_property(JE_IMPORTED_LIBRARY 3rdParty::lz4 IMPORTED_LOCATION)
if(NOT JE_IMPORTED_LIBRARY STREQUAL "${JE_LZ4_PACKAGE_ROOT}/lz4/lib/lz4.lib")
    message(FATAL_ERROR "JE lz4 consumer resolved a different library")
endif()
add_executable(JELz4Consumer Lz4Consumer.cpp)
target_compile_features(JELz4Consumer PRIVATE cxx_std_17)
target_compile_options(JELz4Consumer PRIVATE /W4 /WX)
target_link_libraries(JELz4Consumer PRIVATE 3rdParty::lz4)
enable_testing()
foreach(JE_CASE IN ITEMS blocks frames fragmented errors)
    add_test(NAME JE.LocalPackages.Lz4.${JE_CASE} COMMAND JELz4Consumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.Lz4.${JE_CASE} PROPERTIES TIMEOUT 30)
endforeach()
