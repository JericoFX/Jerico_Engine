cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
foreach(JE_PATH IN ITEMS JE_SAMPLERATE_PACKAGE_ROOT CMAKE_BINARY_DIR)
    JE_python_require_local_path(${JE_PATH})
endforeach()
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_OUTPUT_IN_BUILD)
if(NOT JE_OUTPUT_IN_BUILD OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE libsamplerate consumer output must be beneath this checkout/build")
endif()
project(JELocalLibSamplerateConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE libsamplerate consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_SAMPLERATE_PACKAGE_ROOT}")
find_package(libsamplerate 0.2.1 EXACT REQUIRED MODULE)
add_executable(JELibSamplerateConsumer LibSamplerateConsumer.cpp)
target_compile_features(JELibSamplerateConsumer PRIVATE cxx_std_17)
target_compile_options(JELibSamplerateConsumer PRIVATE /W4 /WX)
target_link_libraries(JELibSamplerateConsumer PRIVATE 3rdParty::libsamplerate)
set_property(TARGET JELibSamplerateConsumer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
enable_testing()
foreach(JE_CASE IN ITEMS api conversions fast medium best reset errors)
    add_test(NAME JE.LocalPackages.LibSamplerate.${JE_CASE} COMMAND JELibSamplerateConsumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.LibSamplerate.${JE_CASE} PROPERTIES TIMEOUT 60)
endforeach()
