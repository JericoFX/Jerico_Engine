cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_POLY2TRI_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_IN_BUILD)
if(NOT JE_IN_BUILD OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE poly2tri consumer build must be beneath this checkout/build")
endif()
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Original static package configurations")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
project(JELocalPoly2TriConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE poly2tri consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_POLY2TRI_PACKAGE_ROOT}")
find_package(poly2tri REQUIRED MODULE)
add_executable(JEPoly2TriConsumer Poly2TriConsumer.cpp)
target_compile_features(JEPoly2TriConsumer PRIVATE cxx_std_17)
target_compile_options(JEPoly2TriConsumer PRIVATE /W4 /WX /EHsc)
target_link_libraries(JEPoly2TriConsumer PRIVATE 3rdParty::poly2tri)
enable_testing()
foreach(JE_CASE IN ITEMS concave hole steiner repeated)
    add_test(NAME JE.LocalPackages.Poly2Tri.${JE_CASE} COMMAND JEPoly2TriConsumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.Poly2Tri.${JE_CASE} PROPERTIES TIMEOUT 20)
endforeach()
