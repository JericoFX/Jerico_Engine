cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_D3DX12_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_BUILD_IN_CHECKOUT)
if(NOT JE_BUILD_IN_CHECKOUT OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE d3dx12 consumer build must be beneath this checkout/build")
endif()
if(NOT EXISTS "${JE_D3DX12_PACKAGE_ROOT}/Findd3dx12.cmake")
    message(FATAL_ERROR "JE_D3DX12_PACKAGE_ROOT must contain the extracted JE Findd3dx12.cmake")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Native SDK consumer configurations" FORCE)
project(JELocalD3dx12Consumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE d3dx12 consumer requires Windows MSVC x64 and its installed SDK")
endif()
message(STATUS "JE d3dx12 SDK: ${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}")
list(PREPEND CMAKE_MODULE_PATH "${JE_D3DX12_PACKAGE_ROOT}")
find_package(d3dx12 REQUIRED MODULE)
get_target_property(JE_IMPORTED_TYPE 3rdParty::d3dx12 TYPE)
get_target_property(JE_IMPORTED_INCLUDE 3rdParty::d3dx12 INTERFACE_INCLUDE_DIRECTORIES)
if(NOT JE_IMPORTED_TYPE STREQUAL "INTERFACE_LIBRARY"
    OR NOT JE_IMPORTED_INCLUDE STREQUAL "${JE_D3DX12_PACKAGE_ROOT}/d3dx12/include")
    message(FATAL_ERROR "JE d3dx12 consumer resolved a different provider")
endif()
add_executable(JED3dx12Consumer D3dx12Consumer.cpp)
target_compile_features(JED3dx12Consumer PRIVATE cxx_std_17)
target_compile_definitions(JED3dx12Consumer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(JED3dx12Consumer PRIVATE /W4 /WX)
target_link_libraries(JED3dx12Consumer PRIVATE 3rdParty::d3dx12)
enable_testing()
foreach(JE_CASE IN ITEMS resources copies tiles)
    add_test(NAME JE.LocalPackages.d3dx12.${JE_CASE} COMMAND JED3dx12Consumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.d3dx12.${JE_CASE} PROPERTIES TIMEOUT 30)
endforeach()
