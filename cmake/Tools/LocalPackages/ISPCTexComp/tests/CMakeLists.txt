cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_ISPC_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_OUTPUT_IN_BUILD)
if(NOT JE_OUTPUT_IN_BUILD OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE ISPCTexComp consumer output must be beneath this checkout/build")
endif()
project(JELocalISPCTexCompConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE ISPCTexComp consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_ISPC_PACKAGE_ROOT}")
find_package(ISPCTexComp REQUIRED MODULE)
if(NOT ISPCTexComp_VERSION_STRING STREQUAL "36b80aa")
    message(FATAL_ERROR "JE ISPCTexComp consumer requires the original pinned source version")
endif()
add_executable(JEISPCTexCompConsumer ISPCTexCompConsumer.cpp)
target_compile_features(JEISPCTexCompConsumer PRIVATE cxx_std_17)
target_compile_options(JEISPCTexCompConsumer PRIVATE /W4 /WX)
target_link_libraries(JEISPCTexCompConsumer PRIVATE 3rdParty::ISPCTexComp)
set_property(TARGET JEISPCTexCompConsumer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_custom_command(TARGET JEISPCTexCompConsumer POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${ISPCTexComp_RUNTIME_DEPENDENCIES}" "$<TARGET_FILE_DIR:JEISPCTexCompConsumer>")
enable_testing()
foreach(JE_CASE IN ITEMS exports bc1 bc3 bc6h bc7 etc1 astc borders)
    add_test(NAME JE.LocalPackages.ISPCTexComp.${JE_CASE} COMMAND JEISPCTexCompConsumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.ISPCTexComp.${JE_CASE} PROPERTIES TIMEOUT 120)
endforeach()
