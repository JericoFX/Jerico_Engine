cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_OPENSSL_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
if(NOT EXISTS "${JE_OPENSSL_PACKAGE_ROOT}/FindOpenSSL.cmake")
    message(FATAL_ERROR "JE_OPENSSL_PACKAGE_ROOT must contain the extracted JE local FindOpenSSL.cmake")
endif()
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_LOCAL_BUILD)
if(NOT JE_LOCAL_BUILD OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE OpenSSL consumer build must be beneath this checkout/build")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Release consumer" FORCE)
project(JELocalOpenSSLConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE OpenSSL consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_OPENSSL_PACKAGE_ROOT}")
find_package(OpenSSL 1.1.1 EXACT REQUIRED MODULE)
if(NOT OPENSSL_VERSION STREQUAL "1.1.1o")
    message(FATAL_ERROR "JE consumer requires exact OpenSSL 1.1.1o")
endif()
get_target_property(JE_SSL_LOCATION OpenSSL::SSL IMPORTED_LOCATION)
if(NOT JE_SSL_LOCATION STREQUAL "${JE_OPENSSL_PACKAGE_ROOT}/OpenSSL/lib/libssl.lib")
    message(FATAL_ERROR "JE consumer resolved a different SSL library")
endif()
add_executable(JEOpenSSLConsumer OpenSSLConsumer.cpp WindowsFirst.cpp OpenSSLFirst.cpp)
target_compile_features(JEOpenSSLConsumer PRIVATE cxx_std_17)
target_compile_options(JEOpenSSLConsumer PRIVATE /W4 /WX)
target_link_libraries(JEOpenSSLConsumer PRIVATE 3rdParty::OpenSSL)
enable_testing()
foreach(JE_CASE IN ITEMS crypto ownership tls azapi static-engines)
    add_test(NAME JE.LocalPackages.OpenSSL.${JE_CASE} COMMAND JEOpenSSLConsumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.OpenSSL.${JE_CASE} PROPERTIES TIMEOUT 60)
endforeach()
# Optional full producer replay provided by its own local output, never the package.
if(DEFINED JE_OPENSSL_VALID_INPUTS)
    JE_python_require_local_path(JE_OPENSSL_VALID_INPUTS)
    add_test(NAME JE.LocalPackages.OpenSSL.Guards COMMAND "${CMAKE_COMMAND}"
        "-DJE_OPENSSL_VALID_INPUTS:FILEPATH=${JE_OPENSSL_VALID_INPUTS}"
        "-DJE_GUARD_OUTPUT:PATH=${CMAKE_BINARY_DIR}/guard-outputs"
        -P "${CMAKE_CURRENT_LIST_DIR}/RunGuardTests.cmake")
    set_tests_properties(JE.LocalPackages.OpenSSL.Guards PROPERTIES TIMEOUT 300)
endif()
