cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_TIFF_PACKAGE_ROOT)
include("${CMAKE_CURRENT_LIST_DIR}/../VerifyZlibInput.cmake")
if(NOT EXISTS "${JE_TIFF_PACKAGE_ROOT}/FindTIFF.cmake")
    message(FATAL_ERROR "JE local TIFF consumer: package root is missing its Find module")
endif()
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
set(JE_CONSUMER_BINARY "${CMAKE_BINARY_DIR}")
JE_python_require_local_path(JE_CONSUMER_BINARY)
cmake_path(IS_PREFIX JE_BUILD_ROOT "${JE_CONSUMER_BINARY}" NORMALIZE JE_BINARY_IN_BUILD)
if(NOT JE_BINARY_IN_BUILD OR JE_CONSUMER_BINARY STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE local TIFF consumer: binary output must be beneath this checkout/build")
endif()
project(JELocalTiffConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE local TIFF consumer requires native Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_TIFF_PACKAGE_ROOT}" "${JE_ZLIB_PACKAGE_ROOT}")
find_package(TIFF 4.2.0.15 EXACT REQUIRED MODULE)
if(NOT TARGET TIFF::TIFF OR NOT TARGET 3rdParty::TIFF OR NOT TARGET ZLIB::ZLIB)
    message(FATAL_ERROR "JE local TIFF consumer: original imported aliases are missing")
endif()
get_target_property(JE_IMPORTED_LIBRARY TIFF::TIFF IMPORTED_LOCATION)
if(NOT JE_IMPORTED_LIBRARY STREQUAL "${JE_TIFF_PACKAGE_ROOT}/tiff/lib/tiff.lib")
    message(FATAL_ERROR "JE local TIFF consumer resolved a different library")
endif()
add_executable(JE_TiffConsumer TiffConsumer.cpp)
target_compile_features(JE_TiffConsumer PRIVATE cxx_std_17)
target_compile_options(JE_TiffConsumer PRIVATE /W4 /WX)
set_target_properties(JE_TiffConsumer PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(JE_TiffConsumer PRIVATE 3rdParty::TIFF)
enable_testing()
foreach(JE_MODE IN ITEMS Identity Codecs Scanlines Tiles Float Errors)
    add_test(NAME JE.LocalPackages.TIFF.${JE_MODE} COMMAND JE_TiffConsumer ${JE_MODE})
endforeach()
add_test(NAME JE.LocalPackages.TIFF.NativeFile COMMAND JE_TiffConsumer NativeFile "${CMAKE_CURRENT_BINARY_DIR}/native-file-roundtrip.tif")
