cmake_minimum_required(VERSION 3.24)
if(NOT JE_ZLIB_PACKAGE_ROOT)
    message(FATAL_ERROR "Set JE_ZLIB_PACKAGE_ROOT to the extracted JE local zlib package")
endif()
file(TO_CMAKE_PATH "${JE_ZLIB_PACKAGE_ROOT}" JE_ZLIB_LOCAL_ROOT)
if(NOT IS_ABSOLUTE "${JE_ZLIB_LOCAL_ROOT}" OR JE_ZLIB_LOCAL_ROOT MATCHES "^//|://")
    message(FATAL_ERROR "JE_ZLIB_PACKAGE_ROOT must be an absolute local directory")
endif()
set(JE_ANCESTOR "${JE_ZLIB_LOCAL_ROOT}")
while(NOT JE_ANCESTOR STREQUAL "")
    if(IS_SYMLINK "${JE_ANCESTOR}")
        message(FATAL_ERROR "JE_ZLIB_PACKAGE_ROOT must not contain symlink paths")
    endif()
    get_filename_component(JE_PARENT "${JE_ANCESTOR}" DIRECTORY)
    if(JE_PARENT STREQUAL JE_ANCESTOR)
        break()
    endif()
    set(JE_ANCESTOR "${JE_PARENT}")
endwhile()
set(JE_ZLIB_PACKAGE_ROOT "${JE_ZLIB_LOCAL_ROOT}")
if(NOT EXISTS "${JE_ZLIB_PACKAGE_ROOT}/FindZLIB.cmake")
    message(FATAL_ERROR "Set JE_ZLIB_PACKAGE_ROOT to the extracted JE local zlib package")
endif()
project(JELocalZlibConsumer LANGUAGES CXX)
list(PREPEND CMAKE_MODULE_PATH "${JE_ZLIB_PACKAGE_ROOT}")
find_package(ZLIB 1.2.11 EXACT REQUIRED MODULE)
if(NOT TARGET 3rdParty::ZLIB OR NOT TARGET ZLIB::ZLIB)
    message(FATAL_ERROR "JE local zlib imported targets are missing")
endif()
get_target_property(JE_IMPORTED_LIBRARY 3rdParty::ZLIB IMPORTED_LOCATION)
if(NOT JE_IMPORTED_LIBRARY STREQUAL "${JE_ZLIB_PACKAGE_ROOT}/zlib/lib/zlibstatic.lib")
    message(FATAL_ERROR "JE local zlib consumer resolved a different library")
endif()
add_executable(JEZlibRoundtrip ZlibRoundtrip.cpp)
target_compile_features(JEZlibRoundtrip PRIVATE cxx_std_17)
target_link_libraries(JEZlibRoundtrip PRIVATE ZLIB::ZLIB)
if(MSVC)
    target_compile_options(JEZlibRoundtrip PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME JE.LocalPackages.ZlibRoundtrip COMMAND JEZlibRoundtrip)
