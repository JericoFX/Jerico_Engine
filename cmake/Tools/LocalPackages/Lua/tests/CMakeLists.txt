cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_LUA_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_CONSUMER_IN_BUILD)
if(NOT JE_CONSUMER_IN_BUILD OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE local Lua: consumer build must be beneath this checkout/build")
endif()
if(NOT EXISTS "${JE_LUA_PACKAGE_ROOT}/FindLua.cmake")
    message(FATAL_ERROR "JE_LUA_PACKAGE_ROOT must contain the extracted JE local FindLua.cmake")
endif()
project(JELocalLuaConsumer LANGUAGES CXX)
list(PREPEND CMAKE_MODULE_PATH "${JE_LUA_PACKAGE_ROOT}")
find_package(Lua 5.4.4 EXACT REQUIRED MODULE)
if(NOT TARGET 3rdParty::Lua)
    message(FATAL_ERROR "JE local Lua imported target is missing")
endif()
get_target_property(JE_IMPORTED_LIBRARY 3rdParty::Lua IMPORTED_LOCATION)
if(NOT JE_IMPORTED_LIBRARY STREQUAL "${JE_LUA_PACKAGE_ROOT}/Lua/lib/release/lualib.lib")
    message(FATAL_ERROR "JE local Lua consumer resolved a different library")
endif()
set(JE_MISSING_DLL "${CMAKE_CURRENT_BINARY_DIR}/JE_Lua_missing_69414532.dll")
if(EXISTS "${JE_MISSING_DLL}")
    message(FATAL_ERROR "Lua error fixture must be an absent local DLL")
endif()
add_executable(JELuaConsumer LuaConsumer.cpp)
target_compile_features(JELuaConsumer PRIVATE cxx_std_17)
target_link_libraries(JELuaConsumer PRIVATE 3rdParty::Lua)
if(MSVC)
    target_compile_options(JELuaConsumer PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME JE.LocalPackages.Lua.CApi COMMAND JELuaConsumer api)
add_test(NAME JE.LocalPackages.Lua.Bytecode COMMAND JELuaConsumer bytecode)
add_test(NAME JE.LocalPackages.Lua.WindowsPatch COMMAND JELuaConsumer patch "${JE_MISSING_DLL}")
