cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_ZSTD_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_BUILD_IS_LOCAL)
if(NOT JE_BUILD_IS_LOCAL OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE zstd consumer build must be beneath this checkout/build")
endif()
if(NOT EXISTS "${JE_ZSTD_PACKAGE_ROOT}/Findzstd.cmake")
    message(FATAL_ERROR "JE_ZSTD_PACKAGE_ROOT must contain the extracted JE local Findzstd.cmake")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Release package consumer" FORCE)
project(JELocalZstdConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE zstd consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_ZSTD_PACKAGE_ROOT}")
find_package(zstd 1.3.5 EXACT REQUIRED MODULE)
get_target_property(JE_IMPORTED_LIBRARY 3rdParty::zstd IMPORTED_LOCATION)
if(NOT JE_IMPORTED_LIBRARY STREQUAL "${JE_ZSTD_PACKAGE_ROOT}/zstd/lib/release/zstd_static.lib")
    message(FATAL_ERROR "JE zstd consumer resolved a different library")
endif()
add_executable(JEZstdConsumer ZstdConsumer.cpp)
target_compile_features(JEZstdConsumer PRIVATE cxx_std_17)
target_compile_options(JEZstdConsumer PRIVATE /W4 /WX)
target_link_libraries(JEZstdConsumer PRIVATE 3rdParty::zstd)
enable_testing()
add_test(NAME JE.LocalPackages.Zstd.Basic COMMAND JEZstdConsumer basic)
add_test(NAME JE.LocalPackages.Zstd.Errors COMMAND JEZstdConsumer errors)
add_test(NAME JE.LocalPackages.Zstd.AdvancedStreams COMMAND JEZstdConsumer streams)
add_test(NAME JE.LocalPackages.Zstd.AllocatorFailure COMMAND JEZstdConsumer allocator)
set_tests_properties(JE.LocalPackages.Zstd.Basic JE.LocalPackages.Zstd.Errors
    JE.LocalPackages.Zstd.AdvancedStreams JE.LocalPackages.Zstd.AllocatorFailure PROPERTIES TIMEOUT 30)
