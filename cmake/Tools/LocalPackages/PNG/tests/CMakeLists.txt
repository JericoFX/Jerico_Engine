cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_PNG_PACKAGE_ROOT)
include("${CMAKE_CURRENT_LIST_DIR}/../VerifyZlibInput.cmake")
if(NOT EXISTS "${JE_PNG_PACKAGE_ROOT}/FindPNG.cmake")
    message(FATAL_ERROR "JE local PNG consumer: package root is missing its Find module")
endif()
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
set(JE_CONSUMER_BINARY "${CMAKE_BINARY_DIR}")
JE_python_require_local_path(JE_CONSUMER_BINARY)
cmake_path(IS_PREFIX JE_BUILD_ROOT "${JE_CONSUMER_BINARY}" NORMALIZE JE_BINARY_IN_BUILD)
if(NOT JE_BINARY_IN_BUILD OR JE_CONSUMER_BINARY STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE local PNG consumer: binary output must be beneath this checkout/build")
endif()
project(JELocalPngConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE local PNG consumer requires native Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_PNG_PACKAGE_ROOT}" "${JE_ZLIB_PACKAGE_ROOT}")
find_package(PNG 1.6.37 EXACT REQUIRED MODULE)
if(NOT TARGET PNG::PNG OR NOT TARGET 3rdParty::PNG OR NOT TARGET ZLIB::ZLIB)
    message(FATAL_ERROR "JE local PNG consumer: original imported aliases are missing")
endif()
get_target_property(JE_IMPORTED_LIBRARY PNG::PNG IMPORTED_LOCATION)
if(NOT JE_IMPORTED_LIBRARY STREQUAL "${JE_PNG_PACKAGE_ROOT}/png/lib/libpng16_static.lib")
    message(FATAL_ERROR "JE local PNG consumer resolved a different library")
endif()
add_executable(JE_PngConsumer PngConsumer.cpp)
target_compile_features(JE_PngConsumer PRIVATE cxx_std_17)
target_compile_options(JE_PngConsumer PRIVATE /W4 /WX)
set_target_properties(JE_PngConsumer PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
target_link_libraries(JE_PngConsumer PRIVATE 3rdParty::PNG)
enable_testing()
foreach(JE_MODE IN ITEMS Identity Roundtrip Transforms Interlace Errors)
    add_test(NAME JE.LocalPackages.PNG.${JE_MODE} COMMAND JE_PngConsumer ${JE_MODE})
endforeach()
