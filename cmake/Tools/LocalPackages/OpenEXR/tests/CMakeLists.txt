cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_OPENEXR_PACKAGE_ROOT)
include("${CMAKE_CURRENT_LIST_DIR}/../VerifyZlibInput.cmake")
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
set(JE_CONSUMER_BINARY "${CMAKE_BINARY_DIR}")
JE_python_require_local_path(JE_CONSUMER_BINARY)
cmake_path(IS_PREFIX JE_BUILD_ROOT "${JE_CONSUMER_BINARY}" NORMALIZE JE_BINARY_IN_BUILD)
if(NOT JE_BINARY_IN_BUILD OR JE_CONSUMER_BINARY STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE local OpenEXR consumer: binary output must be beneath this checkout/build")
endif()
if(NOT EXISTS "${JE_OPENEXR_PACKAGE_ROOT}/FindOpenEXR.cmake")
    message(FATAL_ERROR "JE local OpenEXR consumer: explicit package root is missing")
endif()
project(JELocalOpenEXRConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE local OpenEXR consumer requires native Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_OPENEXR_PACKAGE_ROOT}" "${JE_ZLIB_PACKAGE_ROOT}")
find_package(OpenEXR 3.1.3 EXACT REQUIRED MODULE)
foreach(JE_COMPONENT IN ITEMS OpenEXRConfig IexConfig IlmThreadConfig Iex IlmThread OpenEXRCore OpenEXRUtil OpenEXR)
    if(NOT TARGET 3rdParty::${JE_COMPONENT} OR NOT TARGET OpenEXR::${JE_COMPONENT})
        message(FATAL_ERROR "JE local OpenEXR consumer: original component ${JE_COMPONENT} is missing")
    endif()
endforeach()
foreach(JE_LIBRARY IN ITEMS OpenEXR::Iex OpenEXR::IlmThread OpenEXR::OpenEXRCore OpenEXR::OpenEXRUtil OpenEXR::OpenEXR Imath::Imath)
    get_target_property(JE_TYPE ${JE_LIBRARY} TYPE)
    if(NOT JE_TYPE STREQUAL "STATIC_LIBRARY")
        message(FATAL_ERROR "JE local OpenEXR consumer: ${JE_LIBRARY} must be static")
    endif()
    foreach(JE_CONFIG IN ITEMS DEBUG RELEASE)
        get_target_property(JE_LOCATION ${JE_LIBRARY} IMPORTED_LOCATION_${JE_CONFIG})
        cmake_path(IS_PREFIX JE_OPENEXR_PACKAGE_ROOT "${JE_LOCATION}" NORMALIZE JE_LIBRARY_LOCAL)
        if(NOT JE_LIBRARY_LOCAL OR NOT EXISTS "${JE_LOCATION}")
            message(FATAL_ERROR "JE local OpenEXR consumer: ${JE_LIBRARY}/${JE_CONFIG} is not in this payload")
        endif()
    endforeach()
endforeach()
if(NOT TARGET 3rdParty::Imath OR NOT TARGET 3rdParty::ImathConfig)
    message(FATAL_ERROR "JE local OpenEXR consumer: original Imath aliases are missing")
endif()
add_executable(JE_OpenEXRConsumer OpenEXRConsumer.cpp)
target_compile_features(JE_OpenEXRConsumer PRIVATE cxx_std_17)
target_compile_options(JE_OpenEXRConsumer PRIVATE /W4 /WX /EHsc)
set_target_properties(JE_OpenEXRConsumer PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
target_link_libraries(JE_OpenEXRConsumer PRIVATE 3rdParty::OpenEXR 3rdParty::OpenEXRCore 3rdParty::Imath)
enable_testing()
foreach(JE_MODE IN ITEMS Identity Half Float FillAlpha Errors)
    add_test(NAME JE.LocalPackages.OpenEXR.${JE_MODE} COMMAND JE_OpenEXRConsumer ${JE_MODE})
endforeach()
add_test(NAME JE.LocalPackages.OpenEXR.NativeFile COMMAND JE_OpenEXRConsumer NativeFile "${CMAKE_CURRENT_BINARY_DIR}/consumer-native-$<CONFIG>.exr")
