cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_EXPAT_PACKAGE_ROOT)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_BUILD_IN_CHECKOUT)
if(NOT JE_BUILD_IN_CHECKOUT OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE Expat consumer build must be beneath this checkout/build")
endif()
if(NOT EXISTS "${JE_EXPAT_PACKAGE_ROOT}/Findexpat.cmake")
    message(FATAL_ERROR "JE_EXPAT_PACKAGE_ROOT must contain the extracted local Findexpat.cmake")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Consume Release C archive in both configurations" FORCE)
project(JELocalExpatConsumer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE Expat consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_EXPAT_PACKAGE_ROOT}")
find_package(expat 2.4.2 EXACT REQUIRED MODULE)
get_target_property(JE_IMPORTED_LIBRARY 3rdParty::expat IMPORTED_LOCATION)
if(NOT JE_IMPORTED_LIBRARY STREQUAL "${JE_EXPAT_PACKAGE_ROOT}/expat/lib/libexpatMD.lib"
    OR NOT TARGET expat::expat OR NOT TARGET EXPAT::EXPAT)
    message(FATAL_ERROR "JE Expat consumer resolved a different archive/interface")
endif()
add_executable(JEExpatConsumer ExpatConsumer.cpp)
target_compile_features(JEExpatConsumer PRIVATE cxx_std_17)
target_compile_options(JEExpatConsumer PRIVATE /W4 /WX)
target_link_libraries(JEExpatConsumer PRIVATE 3rdParty::expat)
enable_testing()
foreach(JE_CASE IN ITEMS streaming namespaces errors memory)
    add_test(NAME JE.LocalPackages.Expat.${JE_CASE} COMMAND JEExpatConsumer ${JE_CASE})
    set_tests_properties(JE.LocalPackages.Expat.${JE_CASE} PROPERTIES TIMEOUT 30)
endforeach()
