cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../Python/JERequireLocalPath.cmake")
foreach(JE_PATH IN ITEMS JE_VVL_WORK JE_VVL_INSTALL_ROOT CMAKE_BINARY_DIR)
    JE_python_require_local_path(${JE_PATH})
endforeach()
project(JELocalVulkanValidationLayers LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE VVL requires Windows MSVC x64")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(VulkanHeaders_DIR "${JE_VVL_INSTALL_ROOT}/headers/share/cmake/VulkanHeaders")
set(VulkanUtilityLibraries_DIR "${JE_VVL_INSTALL_ROOT}/utility/lib/cmake/VulkanUtilityLibraries")
set(SPIRV-Headers_DIR "${JE_VVL_INSTALL_ROOT}/spirv-headers/share/cmake/SPIRV-Headers")
set(SPIRV-Tools_DIR "${JE_VVL_INSTALL_ROOT}/spirv-tools/SPIRV-Tools/cmake")
set(SPIRV-Tools-opt_DIR "${JE_VVL_INSTALL_ROOT}/spirv-tools/SPIRV-Tools-opt/cmake")
set(robin_hood_DIR "${JE_VVL_INSTALL_ROOT}/robin/lib/cmake/robin_hood")
set(mimalloc_DIR "${JE_VVL_INSTALL_ROOT}/mimalloc/lib/cmake/mimalloc-2.0")
foreach(JE_CONFIG_DIR IN ITEMS VulkanHeaders_DIR VulkanUtilityLibraries_DIR SPIRV-Headers_DIR
    SPIRV-Tools_DIR SPIRV-Tools-opt_DIR robin_hood_DIR mimalloc_DIR)
    if(NOT IS_DIRECTORY "${${JE_CONFIG_DIR}}")
        message(FATAL_ERROR "JE VVL: explicit local package config is missing: ${JE_CONFIG_DIR}")
    endif()
endforeach()
find_package(VulkanHeaders REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${VulkanHeaders_DIR}")
find_package(VulkanUtilityLibraries REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${VulkanUtilityLibraries_DIR}")
find_package(SPIRV-Headers REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${SPIRV-Headers_DIR}")
find_package(SPIRV-Tools REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${SPIRV-Tools_DIR}")
find_package(SPIRV-Tools-opt REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${SPIRV-Tools-opt_DIR}")
find_package(robin_hood REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${robin_hood_DIR}")
find_package(mimalloc REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${mimalloc_DIR}")
string(TOUPPER "${CMAKE_BUILD_TYPE}" JE_CONFIG)
file(WRITE "${CMAKE_BINARY_DIR}/dependency-exports.txt" "configuration=${JE_CONFIG}\n")
foreach(JE_TARGET IN ITEMS Vulkan::LayerSettings SPIRV-Tools-static SPIRV-Tools-opt mimalloc-static)
    if(NOT TARGET ${JE_TARGET})
        message(FATAL_ERROR "JE VVL: original dependency target is missing: ${JE_TARGET}")
    endif()
    get_target_property(JE_IMPORTED_FILE ${JE_TARGET} IMPORTED_LOCATION_${JE_CONFIG})
    if(NOT JE_IMPORTED_FILE OR NOT EXISTS "${JE_IMPORTED_FILE}")
        message(FATAL_ERROR "JE VVL: ${JE_CONFIG} imported archive missing: ${JE_TARGET}")
    endif()
    JE_python_require_local_path(JE_IMPORTED_FILE)
    cmake_path(IS_PREFIX JE_VVL_INSTALL_ROOT "${JE_IMPORTED_FILE}" NORMALIZE JE_IMPORTED_LOCAL)
    if(NOT JE_IMPORTED_LOCAL)
        message(FATAL_ERROR "JE VVL: imported archive escaped its private install root")
    endif()
    file(APPEND "${CMAKE_BINARY_DIR}/dependency-exports.txt" "${JE_TARGET}=${JE_IMPORTED_FILE}\n")
endforeach()
# SPIRV-Tools always defines a shared auxiliary target in this exact upstream.
# Preserve its private build/install; the validation layer links the static target.
get_target_property(JE_AUXILIARY SPIRV-Tools-shared IMPORTED_LOCATION_${JE_CONFIG})
if(NOT JE_AUXILIARY OR NOT EXISTS "${JE_AUXILIARY}")
    message(FATAL_ERROR "JE VVL: original SPIRV shared auxiliary export is missing")
endif()
JE_python_require_local_path(JE_AUXILIARY)
cmake_path(IS_PREFIX JE_VVL_INSTALL_ROOT "${JE_AUXILIARY}" NORMALIZE JE_AUXILIARY_LOCAL)
if(NOT JE_AUXILIARY_LOCAL)
    message(FATAL_ERROR "JE VVL: shared auxiliary escaped its private install root")
endif()
file(APPEND "${CMAKE_BINARY_DIR}/dependency-exports.txt" "private auxiliary=${JE_AUXILIARY}\n")
set(UPDATE_DEPS OFF CACHE BOOL "All sources are explicitly local" FORCE)
set(VVL_CODEGEN OFF CACHE BOOL "Use the original pre-generated sources" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "Original layer package configuration" FORCE)
set(BUILD_LAYERS ON CACHE BOOL "Preserve the actual validation layer" FORCE)
set(USE_ROBIN_HOOD_HASHING ON CACHE BOOL "Original provided hashing dependency" FORCE)
set(USE_MIMALLOC ON CACHE BOOL "Original Windows x64 allocator dependency" FORCE)
add_subdirectory("${JE_VVL_WORK}" upstream)
