cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../Python/JERequireLocalPath.cmake")
foreach(JE_PATH IN ITEMS JE_VVL_PACKAGE_ROOT JE_VVL_HEADERS_ROOT JE_VVL_LOADER CMAKE_BINARY_DIR)
    JE_python_require_local_path(${JE_PATH})
endforeach()
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
cmake_path(IS_PREFIX JE_BUILD_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE JE_BUILD_LOCAL)
if(NOT JE_BUILD_LOCAL OR CMAKE_BINARY_DIR STREQUAL JE_BUILD_ROOT)
    message(FATAL_ERROR "JE VVL consumer output must be beneath this checkout/build")
endif()
JE_python_verify_file("${JE_VVL_LOADER}" "${JE_VVL_LOADER_SHA256}" "local Vulkan loader")
JE_python_verify_file("${JE_VVL_HEADERS_ROOT}/include/vulkan/vk_layer.h"
    "861cd0ea24c7d81b935e5726520d184dbbc00e571f9089e17a723ccbbc1944e2" "audited Vulkan 1.3.261 loader-layer header")
project(JELocalVulkanValidationLayersConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE VVL consumer requires Windows MSVC x64")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_VVL_PACKAGE_ROOT}")
find_package(vulkan-validationlayers 1.3.261 EXACT REQUIRED MODULE)
add_executable(JEVulkanValidationLayersConsumer VulkanValidationLayersConsumer.cpp)
target_compile_features(JEVulkanValidationLayersConsumer PRIVATE cxx_std_17)
target_compile_options(JEVulkanValidationLayersConsumer PRIVATE /W4 /WX)
target_include_directories(JEVulkanValidationLayersConsumer SYSTEM PRIVATE "${JE_VVL_HEADERS_ROOT}/include")
target_link_libraries(JEVulkanValidationLayersConsumer PRIVATE 3rdParty::vulkan-validationlayers)
enable_testing()
set(JE_EMPTY_IMPLICIT_LAYER_PATH "${CMAKE_CURRENT_BINARY_DIR}/je-vvl-empty-implicit-layers")
JE_python_require_local_path(JE_EMPTY_IMPLICIT_LAYER_PATH)
file(MAKE_DIRECTORY "${JE_EMPTY_IMPLICIT_LAYER_PATH}")
file(GLOB JE_IMPLICIT_LAYER_CONTENTS LIST_DIRECTORIES true "${JE_EMPTY_IMPLICIT_LAYER_PATH}/*")
if(JE_IMPLICIT_LAYER_CONTENTS)
    message(FATAL_ERROR "JE VVL consumer implicit layer directory must be empty")
endif()
foreach(JE_CONFIG IN ITEMS release debug)
    set(JE_PAIR "${JE_VVL_PACKAGE_ROOT}/vulkan-validationlayers/lib/${JE_CONFIG}")
    foreach(JE_CASE IN ITEMS exports runtime)
        add_test(NAME JE.LocalPackages.VVL.${JE_CONFIG}.${JE_CASE}
            COMMAND JEVulkanValidationLayersConsumer ${JE_CASE}
                "${JE_PAIR}/VkLayer_khronos_validation.dll" "${JE_VVL_LOADER}")
        set_tests_properties(JE.LocalPackages.VVL.${JE_CONFIG}.${JE_CASE} PROPERTIES TIMEOUT 60)
        if(JE_CASE STREQUAL "runtime")
            set_tests_properties(JE.LocalPackages.VVL.${JE_CONFIG}.${JE_CASE} PROPERTIES
                ENVIRONMENT "VK_IMPLICIT_LAYER_PATH=${JE_EMPTY_IMPLICIT_LAYER_PATH}")
        endif()
    endforeach()
endforeach()
