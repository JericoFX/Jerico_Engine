cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../Python/JERequireLocalPath.cmake")
JE_python_require_local_path(JE_MCPP_WORK)
JE_python_require_local_path(CMAKE_BINARY_DIR)
get_filename_component(JE_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(JE_BUILD_ROOT "${JE_ENGINE_ROOT}/build")
foreach(JE_PATH IN ITEMS JE_MCPP_WORK CMAKE_BINARY_DIR)
    cmake_path(IS_PREFIX JE_BUILD_ROOT "${${JE_PATH}}" NORMALIZE JE_PATH_IN_BUILD)
    if(NOT JE_PATH_IN_BUILD OR "${${JE_PATH}}" STREQUAL JE_BUILD_ROOT)
        message(FATAL_ERROR "JE local mcpp: ${JE_PATH} must be beneath this checkout/build")
    endif()
endforeach()
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Original mcpp Windows package is Release only" FORCE)
# The original CLI nmake invocation has no optimization/NDEBUG switches; only
# the DLL invocation adds -O2 -DNDEBUG. Preserve these separately below.
set(CMAKE_C_FLAGS_RELEASE "" CACHE STRING "Use the original per-target Release switches" FORCE)
project(JELocalMcpp LANGUAGES C)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JE local mcpp requires Windows MSVC x64")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/McppPatchManifest.json" JE_MANIFEST)
string(JSON JE_PATCHED_COUNT LENGTH "${JE_MANIFEST}" files)
math(EXPR JE_PATCHED_LAST "${JE_PATCHED_COUNT} - 1")
foreach(JE_INDEX RANGE ${JE_PATCHED_LAST})
    string(JSON JE_FILE GET "${JE_MANIFEST}" files ${JE_INDEX} path)
    string(JSON JE_HASH GET "${JE_MANIFEST}" files ${JE_INDEX} patchedSHA256)
    JE_python_verify_file("${JE_MCPP_WORK}/${JE_FILE}" "${JE_HASH}" "patched ${JE_FILE}")
endforeach()
set(JE_SOURCES)
foreach(JE_FILE IN ITEMS main.c directive.c eval.c expand.c support.c system.c mbchar.c)
    list(APPEND JE_SOURCES "${JE_MCPP_WORK}/src/${JE_FILE}")
endforeach()
add_library(JEMcppDll SHARED ${JE_SOURCES})
add_executable(JEMcppCli ${JE_SOURCES})
foreach(JE_TARGET IN ITEMS JEMcppDll JEMcppCli)
    target_compile_definitions(${JE_TARGET} PRIVATE COMPILER=MSC _CRT_SECURE_NO_DEPRECATE)
    target_compile_options(${JE_TARGET} PRIVATE /Za /favor:AMD64 /TC)
    target_link_options(${JE_TARGET} PRIVATE /MACHINE:X64)
endforeach()
target_compile_definitions(JEMcppDll PRIVATE MCPP_LIB=1 DLL_EXPORT=1 NDEBUG)
target_compile_options(JEMcppDll PRIVATE /O2)
set_target_properties(JEMcppDll PROPERTIES OUTPUT_NAME mcpp0 MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set_target_properties(JEMcppCli PROPERTIES OUTPUT_NAME mcpp MSVC_RUNTIME_LIBRARY MultiThreaded)
