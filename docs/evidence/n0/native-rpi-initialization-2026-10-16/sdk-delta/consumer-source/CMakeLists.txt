set(je_targets JE::EditorMaterials JE::EditorCamera JE::EditorRenderIntegration JE::RenderAssetBuilders AZ::SceneData JE::RenderIntegration)
cmake_path(SET je_rpi_delta_root NORMALIZE "${JE_SDK_PREFIX}/../sdk-rpi98-delta")
if(NOT PROJECT_NAME)
cmake_minimum_required(VERSION 3.25)
if(NOT JE_SDK_PREFIX OR NOT EXISTS "${JE_SDK_PREFIX}/cmake/Findo3de.cmake")
    message(FATAL_ERROR "JE_SDK_PREFIX must name the installed SDK")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_SDK_PREFIX}/cmake")
find_package(o3de REQUIRED MODULE)
get_property(je_engine_root GLOBAL PROPERTY O3DE_ENGINE_ROOT_FOLDER)
cmake_path(SET je_expected_root NORMALIZE "${JE_SDK_PREFIX}")
cmake_path(SET je_actual_root NORMALIZE "${je_engine_root}")
string(REGEX REPLACE "/$" "" je_expected_root "${je_expected_root}")
string(REGEX REPLACE "/$" "" je_actual_root "${je_actual_root}")
cmake_path(COMPARE "${je_expected_root}" EQUAL "${je_actual_root}" je_root_matches)
if(NOT je_root_matches)
    message(FATAL_ERROR "Engine resolved outside installed SDK: ${je_engine_root}")
endif()
project(JENativeSdkConsumer LANGUAGES C CXX VERSION 1.0.0)
o3de_initialize()
foreach(je_target IN LISTS je_targets)
    if(NOT TARGET ${je_target})
        message(FATAL_ERROR "Missing native SDK target: ${je_target}")
    endif()
    get_target_property(je_actual_target ${je_target} ALIASED_TARGET)
    if(NOT je_actual_target)
        set(je_actual_target ${je_target})
    endif()
    get_target_property(je_imported ${je_actual_target} IMPORTED)
    if(NOT je_imported)
        message(FATAL_ERROR "Native SDK target is not imported: ${je_target}")
    endif()
    foreach(je_property IMPORTED_LOCATION_PROFILE IMPORTED_IMPLIB_PROFILE)
        get_target_property(je_location ${je_actual_target} ${je_property})
        if(NOT je_location OR NOT EXISTS "${je_location}")
            message(FATAL_ERROR "Missing ${je_target} ${je_property}: ${je_location}")
        endif()
        cmake_path(IS_PREFIX je_expected_root "${je_location}" NORMALIZE je_in_sdk)
        if(NOT je_in_sdk)
            message(FATAL_ERROR "Import path outside SDK: ${je_target} ${je_location}")
        endif()
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-${je_actual_target}-$<CONFIG>.txt"
        CONTENT "target=${je_target}\nimported=${je_imported}\nlocation=$<TARGET_FILE:${je_target}>\nimplib=$<TARGET_LINKER_FILE:${je_target}>\nincludes=$<TARGET_PROPERTY:${je_target},INTERFACE_INCLUDE_DIRECTORIES>\nlinks=$<TARGET_PROPERTY:${je_target},INTERFACE_LINK_LIBRARIES>\n")
endforeach()
# Override only the two reviewed Profile providers; all six targets otherwise remain the installed SDK targets.
foreach(je_provider RenderIntegration EditorRenderIntegration)
    foreach(je_payload "${je_rpi_delta_root}/bin/${je_provider}.dll" "${je_rpi_delta_root}/lib/${je_provider}.lib")
        if(NOT EXISTS "${je_payload}")
            message(FATAL_ERROR "Missing reviewed native RPI delta payload: ${je_payload}")
        endif()
    endforeach()
    set_target_properties(${je_provider} PROPERTIES
        IMPORTED_LOCATION_PROFILE "${je_rpi_delta_root}/bin/${je_provider}.dll"
        IMPORTED_IMPLIB_PROFILE "${je_rpi_delta_root}/lib/${je_provider}.lib")
endforeach()
# Runtime body-only repair: retain the SDK96 import library after the parent ABI/dependency comparison.
get_target_property(je_rpi_public_imported Atom_RPI.Public IMPORTED)
get_target_property(je_rpi_public_implib Atom_RPI.Public IMPORTED_IMPLIB_PROFILE)
if(NOT je_rpi_public_imported OR NOT EXISTS "${je_rpi_public_implib}" OR NOT EXISTS "${je_rpi_delta_root}/bin/Atom_RPI.Public.dll")
    message(FATAL_ERROR "Reviewed Atom_RPI.Public body provider or installed import library is absent")
endif()
cmake_path(SET je_rpi_public_implib_normalized NORMALIZE "${je_rpi_public_implib}")
cmake_path(SET je_rpi_public_expected_implib NORMALIZE "${JE_SDK_PREFIX}/lib/Windows/profile/Default/Atom_RPI.Public.lib")
if(NOT je_rpi_public_implib_normalized STREQUAL je_rpi_public_expected_implib)
    message(FATAL_ERROR "Atom_RPI.Public import library must remain the installed SDK96 library")
endif()
set_target_properties(Atom_RPI.Public PROPERTIES IMPORTED_LOCATION_PROFILE "${je_rpi_delta_root}/bin/Atom_RPI.Public.dll")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rpi-body-provider-$<CONFIG>.txt"
    CONTENT "target=Gem::Atom_RPI.Public\nimported=${je_rpi_public_imported}\nlocation=$<TARGET_FILE:Atom_RPI.Public>\nimplib=$<TARGET_LINKER_FILE:Atom_RPI.Public>\n")
foreach(je_header JE/Render/Integration/RenderIntegrationModule.h JE/Editor/RenderIntegration/RenderIntegrationModule.h)
    if(NOT EXISTS "${je_rpi_delta_root}/include/${je_header}")
        message(FATAL_ERROR "Missing reviewed native RPI public header: ${je_header}")
    endif()
endforeach()
foreach(je_legacy Gem::MaterialEditor Gem::MaterialCanvas Gem::PassCanvas Gem::AtomShader.Builders Gem::AtomShader.Static Gem::Atom_RPI.Builders AtomShader.Builders AtomShader.Static Atom_RPI.Builders Gem::Atom_Bootstrap Gem::Atom_Bootstrap.Headers Gem::Atom_Bootstrap.Clients Gem::Atom_Bootstrap.Servers Gem::Atom_Bootstrap.Unified Atom_Bootstrap Atom_Bootstrap.Headers)
    if(TARGET ${je_legacy})
        message(FATAL_ERROR "Retired standalone, builder or Bootstrap target imported: ${je_legacy}")
    endif()
endforeach()
else()
ly_add_target(
    NAME JENativeSdkConsumer
    NAMESPACE JE
    EXECUTABLE
    NO_UNITY
    FILES_CMAKE consumer_files.cmake
    BUILD_DEPENDENCIES PRIVATE ${je_targets}
)
target_include_directories(JENativeSdkConsumer BEFORE PRIVATE "${je_rpi_delta_root}/include")
endif()
