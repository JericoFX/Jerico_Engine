set(je_targets JE::EditorMaterials JE::EditorCamera JE::EditorRenderIntegration JE::RenderAssetBuilders AZ::SceneData)
if(NOT PROJECT_NAME)
cmake_minimum_required(VERSION 3.25)
if(NOT JE_SDK_PREFIX OR NOT EXISTS "${JE_SDK_PREFIX}/cmake/Findo3de.cmake")
    message(FATAL_ERROR "JE_SDK_PREFIX must name the installed SDK")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_SDK_PREFIX}/cmake")
find_package(o3de REQUIRED MODULE)
get_property(je_engine_root GLOBAL PROPERTY O3DE_ENGINE_ROOT_FOLDER)
cmake_path(SET je_expected_root NORMALIZE "${JE_SDK_PREFIX}")
cmake_path(SET je_actual_root NORMALIZE "${je_engine_root}")
string(REGEX REPLACE "/$" "" je_expected_root "${je_expected_root}")
string(REGEX REPLACE "/$" "" je_actual_root "${je_actual_root}")
cmake_path(COMPARE "${je_expected_root}" EQUAL "${je_actual_root}" je_root_matches)
if(NOT je_root_matches)
    message(FATAL_ERROR "Engine resolved outside installed SDK: ${je_engine_root}")
endif()
project(JENativeSdkConsumer LANGUAGES C CXX VERSION 1.0.0)
o3de_initialize()
foreach(je_target IN LISTS je_targets)
    if(NOT TARGET ${je_target})
        message(FATAL_ERROR "Missing native SDK target: ${je_target}")
    endif()
    get_target_property(je_actual_target ${je_target} ALIASED_TARGET)
    if(NOT je_actual_target)
        set(je_actual_target ${je_target})
    endif()
    get_target_property(je_imported ${je_actual_target} IMPORTED)
    if(NOT je_imported)
        message(FATAL_ERROR "Native SDK target is not imported: ${je_target}")
    endif()
    foreach(je_property IMPORTED_LOCATION_PROFILE IMPORTED_IMPLIB_PROFILE)
        get_target_property(je_location ${je_actual_target} ${je_property})
        if(NOT je_location OR NOT EXISTS "${je_location}")
            message(FATAL_ERROR "Missing ${je_target} ${je_property}: ${je_location}")
        endif()
        cmake_path(IS_PREFIX je_expected_root "${je_location}" NORMALIZE je_in_sdk)
        if(NOT je_in_sdk)
            message(FATAL_ERROR "Import path outside SDK: ${je_target} ${je_location}")
        endif()
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-${je_actual_target}-$<CONFIG>.txt"
        CONTENT "target=${je_target}\nimported=${je_imported}\nlocation=$<TARGET_FILE:${je_target}>\nimplib=$<TARGET_LINKER_FILE:${je_target}>\nincludes=$<TARGET_PROPERTY:${je_target},INTERFACE_INCLUDE_DIRECTORIES>\nlinks=$<TARGET_PROPERTY:${je_target},INTERFACE_LINK_LIBRARIES>\n")
endforeach()
foreach(je_legacy Gem::MaterialEditor Gem::MaterialCanvas Gem::PassCanvas Gem::AtomShader.Builders Gem::AtomShader.Static Gem::Atom_RPI.Builders AtomShader.Builders AtomShader.Static Atom_RPI.Builders)
    if(TARGET ${je_legacy})
        message(FATAL_ERROR "Retired standalone or builder target imported: ${je_legacy}")
    endif()
endforeach()
else()
ly_add_target(
    NAME JENativeSdkConsumer
    NAMESPACE JE
    EXECUTABLE
    NO_UNITY
    FILES_CMAKE consumer_files.cmake
    BUILD_DEPENDENCIES PRIVATE ${je_targets}
)
endif()
