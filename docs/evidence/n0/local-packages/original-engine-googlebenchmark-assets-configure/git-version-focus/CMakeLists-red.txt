cmake_minimum_required(VERSION 3.24)
project(GitVersionRed VERSION 1.7.0 LANGUAGES NONE)
if(NOT DEFINED GET_GIT_MODULE OR NOT DEFINED FAKE_CMAKE OR NOT DEFINED FAKE_GIT_SCRIPT)
  message(FATAL_ERROR "probe inputs missing")
endif()
set(CMAKE_DISABLE_FIND_PACKAGE_Git ON)
set(GIT_EXECUTABLE "${FAKE_CMAKE};-P;${FAKE_GIT_SCRIPT}")
include("${GET_GIT_MODULE}")
get_git_version(GIT_VERSION)
if(NOT EXISTS "$ENV{JE_FAKE_GIT_LOG}")
  message(FATAL_ERROR "RED did not exercise inherited GIT_EXECUTABLE")
endif()
file(STRINGS "$ENV{JE_FAKE_GIT_LOG}" _calls)
list(LENGTH _calls _call_count)
if(NOT _call_count EQUAL 3)
  message(FATAL_ERROR "expected exactly three FakeGit queries, got ${_call_count}: ${_calls}")
endif()
message(STATUS "FOCUS_RED_CONFIRMED: disabling FindGit still ran inherited FakeGit ${_call_count} times; GIT_VERSION=${GIT_VERSION}")
