cmake_minimum_required(VERSION 3.24)
project(benchmark VERSION 1.7.0 LANGUAGES NONE)
if(NOT DEFINED GET_GIT_MODULE OR NOT DEFINED FAKE_CMAKE OR NOT DEFINED FAKE_GIT_SCRIPT)
  message(FATAL_ERROR "probe inputs missing")
endif()
set(_expected_git "${FAKE_CMAKE};-P;${FAKE_GIT_SCRIPT}")
set(GIT_EXECUTABLE "${_expected_git}" CACHE STRING "inherited fake command")
set(CMAKE_DISABLE_FIND_PACKAGE_Git OFF)
block()
  # These two sets are copied from the frozen FindGoogleBenchmark.cmake wrapper.
  set(CMAKE_DISABLE_FIND_PACKAGE_Git ON)
  set(GIT_EXECUTABLE "")
  include("${GET_GIT_MODULE}")
  get_git_version(GIT_VERSION)
  if(NOT "${GIT_VERSION}" STREQUAL "0.0.0")
    message(FATAL_ERROR "empty GIT_EXECUTABLE should produce 0.0.0, got ${GIT_VERSION}")
  endif()
  if("${GIT_VERSION}" STREQUAL "0.0.0")
    set(VERSION "${benchmark_VERSION}")
  else()
    set(VERSION "${GIT_VERSION}")
  endif()
  set(_probe_version "${VERSION}")
endblock()
if(NOT "${_probe_version}" STREQUAL "1.7.0")
  message(FATAL_ERROR "archive fallback was not benchmark_VERSION=1.7.0: ${_probe_version}")
endif()
if(NOT "${GIT_EXECUTABLE}" STREQUAL "${_expected_git}" OR NOT "$CACHE{GIT_EXECUTABLE}" STREQUAL "${_expected_git}")
  message(FATAL_ERROR "outer/cache GIT_EXECUTABLE did not survive block")
endif()
if(CMAKE_DISABLE_FIND_PACKAGE_Git)
  message(FATAL_ERROR "outer FindGit disable flag was not restored")
endif()
if(EXISTS "$ENV{JE_FAKE_GIT_LOG}")
  message(FATAL_ERROR "GREEN unexpectedly executed FakeGit")
endif()
message(STATUS "FOCUS_GREEN_CONFIRMED: GetGitVersion returned 0.0.0, fallback=${_probe_version}; outer/cache GIT_EXECUTABLE preserved; FakeGit calls=0")
