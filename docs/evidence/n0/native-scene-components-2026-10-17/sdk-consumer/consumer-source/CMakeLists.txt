cmake_minimum_required(VERSION 3.25)
if(NOT JE_SDK_PREFIX OR NOT EXISTS "${JE_SDK_PREFIX}/cmake/Findo3de.cmake")
    message(FATAL_ERROR "JE_SDK_PREFIX must name the installed SDK")
endif()
list(PREPEND CMAKE_MODULE_PATH "${JE_SDK_PREFIX}/cmake")
find_package(o3de REQUIRED MODULE)
get_property(je_engine_root GLOBAL PROPERTY O3DE_ENGINE_ROOT_FOLDER)
cmake_path(SET je_expected_root NORMALIZE "${JE_SDK_PREFIX}")
cmake_path(SET je_actual_root NORMALIZE "${je_engine_root}")
string(REGEX REPLACE "/$" "" je_expected_root "${je_expected_root}")
string(REGEX REPLACE "/$" "" je_actual_root "${je_actual_root}")
cmake_path(COMPARE "${je_expected_root}" EQUAL "${je_actual_root}" je_root_matches)
if(NOT je_root_matches)
    message(FATAL_ERROR "Engine resolved outside installed SDK: ${je_engine_root}")
endif()
project(JENativeSdkConsumer LANGUAGES C CXX VERSION 1.0.0)
o3de_initialize()
foreach(je_target IN LISTS je_targets)
    if(NOT TARGET ${je_target})
        message(FATAL_ERROR "Missing native SDK target: ${je_target}")
    endif()
    get_target_property(je_actual_target ${je_target} ALIASED_TARGET)
    if(NOT je_actual_target)
        set(je_actual_target ${je_target})
    endif()
    get_target_property(je_imported ${je_actual_target} IMPORTED)
    if(NOT je_imported)
        message(FATAL_ERROR "Native SDK target is not imported: ${je_target}")
    endif()
    foreach(je_property IMPORTED_LOCATION_PROFILE IMPORTED_IMPLIB_PROFILE)
        get_target_property(je_location ${je_actual_target} ${je_property})
        if(NOT je_location OR NOT EXISTS "${je_location}")
            message(FATAL_ERROR "Missing ${je_target} ${je_property}: ${je_location}")
        endif()
        cmake_path(IS_PREFIX je_expected_root "${je_location}" NORMALIZE je_in_sdk)
        if(NOT je_in_sdk)
            message(FATAL_ERROR "Import path outside SDK: ${je_target} ${je_location}")
        endif()
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-${je_actual_target}-$<CONFIG>.txt"
        CONTENT "target=${je_target}\nimported=${je_imported}\nlocation=$<TARGET_FILE:${je_target}>\nimplib=$<TARGET_LINKER_FILE:${je_target}>\nincludes=$<TARGET_PROPERTY:${je_target},INTERFACE_INCLUDE_DIRECTORIES>\nlinks=$<TARGET_PROPERTY:${je_target},INTERFACE_LINK_LIBRARIES>\n")
endforeach()
# Scene contracts are existing native HEADERONLY/STATIC owners, not shared imports.
foreach(je_target IN LISTS je_scene_targets)
    if(NOT TARGET ${je_target})
        message(FATAL_ERROR "Missing native Scene SDK target: ${je_target}")
    endif()
    get_target_property(je_actual_target ${je_target} ALIASED_TARGET)
    if(NOT je_actual_target)
        set(je_actual_target ${je_target})
    endif()
    get_target_property(je_imported ${je_actual_target} IMPORTED)
    get_target_property(je_type ${je_actual_target} TYPE)
    if(NOT je_imported)
        message(FATAL_ERROR "Scene SDK target is not imported: ${je_target}")
    endif()
    get_target_property(je_includes ${je_actual_target} INTERFACE_INCLUDE_DIRECTORIES)
    if(NOT je_includes)
        message(FATAL_ERROR "Scene public include root missing: ${je_target}")
    endif()
    # Validate concrete Scene include paths from generated reports after generation.
    if(je_target STREQUAL "JE::RenderScene.Public")
        if(NOT je_type STREQUAL "INTERFACE_LIBRARY")
            message(FATAL_ERROR "Scene.Public must remain HEADERONLY")
        endif()
        get_target_property(je_links ${je_actual_target} INTERFACE_LINK_LIBRARIES)
        if(je_links MATCHES "(^|;)(JE::)?RenderScene(;|$)")
            message(FATAL_ERROR "Scene.Public links back to its implementation")
        endif()
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/scene-public-$<CONFIG>.txt"
            CONTENT "target=${je_target}\nimported=${je_imported}\ntype=${je_type}\nincludes=$<TARGET_GENEX_EVAL:${je_actual_target},$<TARGET_PROPERTY:${je_actual_target},INTERFACE_INCLUDE_DIRECTORIES>>\nlinks=${je_links}\n")
    else()
        get_target_property(je_location ${je_actual_target} IMPORTED_LOCATION_PROFILE)
        cmake_path(IS_PREFIX je_expected_root "${je_location}" NORMALIZE je_in_sdk)
        if(NOT je_type STREQUAL "STATIC_LIBRARY" OR NOT je_in_sdk OR NOT EXISTS "${je_location}" OR NOT je_location MATCHES "[.]lib$")
            message(FATAL_ERROR "Invalid installed Scene static library: ${je_target} ${je_location}")
        endif()
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/scene-${je_actual_target}-$<CONFIG>.txt"
            CONTENT "target=${je_target}\nimported=${je_imported}\ntype=${je_type}\nlocation=$<TARGET_FILE:${je_target}>\nincludes=$<TARGET_GENEX_EVAL:${je_actual_target},$<TARGET_PROPERTY:${je_actual_target},INTERFACE_INCLUDE_DIRECTORIES>>\nlinks=$<TARGET_PROPERTY:${je_target},INTERFACE_LINK_LIBRARIES>\n")
    endif()
endforeach()
